fn transform(nested: ((i64, i64), f64)) -> i64 {
    let r1 = 0..=10
    let elem = nested.0.1
    let scaled = 1.5e-3
    let combined = (elem `plus-weighted` 4) |> double
    let mapped = with_resource(combined) do |res| {
        res + 1
    }
    when {
        mapped < 0 => -1,
        else => mapped,
    }
}
