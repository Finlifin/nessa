fn run_with_log() -> i64 handles(read_line) {
    handles(read_line) let cap = make_logger()
    handles read_line(prompt) -> String => {
        resume "ok"
    }
    let text = ask()# ! {
        error IoError.NotFound => "missing",
        catch other => "fallback",
    }
    let applied = work.use(cap).as(i64)
    applied
}

fn handle_stream(fut: Task) -> i64 {
    let v = producer() # {
        emit_item(item, k) => k(item + 1),
    }
    let done = fut.await!
    let r = reset tag {
        shift tag, cont {
            cont(v + done)
        }
    }
    r
}
