fn classify_value(input: Any) -> i64 {
    let is_circle = input matches Shape.circle(r) if r > 0.0
    input match {
        [first: i64, ...middle, last: i64] if first < last => first + last,
        n: i64 and n + 1 is next_val as bound => bound,
        Shape.point or not Shape.rect(_, _) => 0,
        { tag: .ok, count } => count,
        ok! => ok,
        error IoError.* as err => -1,
        _ => 1,
    }
}
