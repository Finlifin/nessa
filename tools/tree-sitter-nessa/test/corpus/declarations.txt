    assoc Item: Type = i64
    def fn len(self) -> usize
    derive fn is_empty(self) -> bool = self.len() == 0
}

    assoc Item = Any
    pub fn len(self) -> usize {
        0
    }
}

    fn push_all(self, .prefix: i64 = 0, ...items: List) -> Unit {
        ()
